# Instantiate the galaxy population
galpop = GalaxyPopulation(
    minimum_stellar_mass=1E9*Msun,
    maximum_stellar_mass=1E11*Msun,
    volume=volume,
    model=model,
    grid=grid,
    cosmology=cosmo,
    redshift=0.0,
    random_seed=42)

print(galpop)

    grid=grid,
    dust_curve=PowerLaw(slope=-1),
    fesc=1E-10,)
